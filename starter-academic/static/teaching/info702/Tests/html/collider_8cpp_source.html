18#include <cmath>
19#include <QtWidgets>
20#include "objects.hpp"
21
22static const int AsteroidCount = 10;
23
24int main(int argc, char **argv)
25{
26 // Initializes Qt.
27 QApplication app(argc, argv);
28 // QRandomGenerator::global() is automatically seeded.
29
30 // Creates a graphics scene where we will put graphical objects.
31 QGraphicsScene graphical_scene;
32 graphical_scene.setSceneRect(0, 0, IMAGE_SIZE, IMAGE_SIZE);
33 graphical_scene.setItemIndexMethod(QGraphicsScene::NoIndex);
34
35 // We choose to check intersection with 100 trials, with up to 100 points per shape.
36 LogicalScene local_scene( 100 );
37 logical_scene = &local_scene;
38
39 // Creates a few asteroids...
40 for (int i = 0; i < AsteroidCount; ++i) {
41 QColor cok( 150, 130, 110 );
42 QColor cko( 255, 240, 0 );
43
44 // A master shape gathers all the elements of the shape.
45 MasterShape* asteroid = new Asteroid( cok, cko,
46 ( QRandomGenerator::global()->bounded( 20 ) + 20 ) / 10.0 /* speed */,
47 (double) (10 + QRandomGenerator::global()->bounded( 40 )) /* radius */ );
48 // Set direction and position
49 asteroid->setRotation(QRandomGenerator::global()->bounded( 360 ));
50 asteroid->setPos( IMAGE_SIZE/2 + ::sin((i * 6.28) / AsteroidCount) * 200,
51 IMAGE_SIZE/2 + ::cos((i * 6.28) / AsteroidCount) * 200 );
52 // Add it to the graphical scene
53 graphical_scene.addItem( asteroid );
54 // and to the logical scene
55 logical_scene->formes.push_back( asteroid );
56 }
57
58
59 // Standard stuff to initialize a graphics view with some background.
60 QGraphicsView view(&graphical_scene);
61 view.setRenderHint(QPainter::Antialiasing);
62 view.setBackgroundBrush(QPixmap(":/images/stars.jpg"));
63 view.setCacheMode(QGraphicsView::CacheBackground);
64 view.setViewportUpdateMode(QGraphicsView::BoundingRectViewportUpdate);
65 view.setDragMode(QGraphicsView::NoDrag); // QGraphicsView::ScrollHandDrag
66 view.setWindowTitle(QT_TRANSLATE_NOOP(QGraphicsView, "Space - the final frontier"));
67 view.setHorizontalScrollBarPolicy ( Qt::ScrollBarAlwaysOff );
68 view.setVerticalScrollBarPolicy ( Qt::ScrollBarAlwaysOff );
69 view.resize( IMAGE_SIZE, IMAGE_SIZE );
70 view.show();
71
72 // Creates a timer that will call `advance()` method regularly.
73 QTimer timer;
74 QObject::connect(&timer, &QTimer::timeout,
75 &graphical_scene, &QGraphicsScene::advance);
76 timer.start( 30 ); // every 30ms
77
78 return app.exec();
79}
