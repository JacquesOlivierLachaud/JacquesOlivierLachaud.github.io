cmake_minimum_required(VERSION 3.16)
project(TPImages LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
function(tp_program target)
  add_executable(${target} ${ARGN})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
tp_program(exemples-cpp20 exemples-cpp20.cpp)
enable_testing()
add_test(NAME exemples-cpp20 COMMAND exemples-cpp20)
