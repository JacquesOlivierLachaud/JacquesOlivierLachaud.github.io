48/// @brief Polymorphic class that represents the top class of any complex
49/// shape.
50///
51/// It takes care of memorizing collisions and storing the
52/// current main coordinates of a shape.
53struct MasterShape : public GraphicalShape
54{
55 enum State { Ok, Collision };
56 MasterShape( QColor cok, QColor cko );
57 /// Set once during construction, before adding this item to a scene.
58 void setGraphicalShape( GraphicalShape* f );
59 virtual void paint( QPainter *painter, const QStyleOptionGraphicsItem *option,
60 QWidget *widget) override;
61 virtual QPointF randomPoint() const override;
62 virtual bool isInside( const QPointF& p ) const override;
63 virtual QRectF boundingRect() const override;
64
65 // Checks if this shape collides with another shape and forces also
66 // the shapes to stay in the graphical view.
67 virtual void advance(int step) override;
68 State currentState() const;
69 QColor currentColor() const;
70
71protected:
72 // Helpers specific to moving master shapes; not part of GraphicalShape.
73 QPointF toLocal( QPointF p ) const;
74 QPointF toParent( QPointF p ) const;
75 GraphicalShape* _f;
76 State _state;
77 QColor _cok, _cko;
78};

107/// @brief A class to store master shapes and to test their possible
108/// collisions with a randomized algorithm.
109struct LogicalScene {
110 std::vector< MasterShape*> formes;
111 int nb_tested;
112
113 /// Builds a logical scene where collisions are detected by checking
114 /// \a n random points within shapes.
115 ///
116 /// @param n any positive integer.
117 LogicalScene( int n );
118 /// Given two shapes \a f1 and \a f2, returns if they collide.
119 /// @param f1 any master shape.
120 /// @param f2 any different master shape.
121 /// @return 'true' if a collision was detected; false negatives are possible.
122 bool intersect( MasterShape* f1, MasterShape* f2 );
123 /// @param f1 any master shape.
124 /// @return 'true' if a collision with another stored master shape was detected.
125 bool intersect( MasterShape* f1 );
126};
