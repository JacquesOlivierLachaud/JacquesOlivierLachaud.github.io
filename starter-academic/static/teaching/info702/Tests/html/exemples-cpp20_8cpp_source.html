1// Vérification autonome des mécanismes C++20 utilisés dans l'énoncé.
2// Les vecteurs représentent ici les séquences de pixels, sans fournir le corrigé d'Image2D.

13struct Color {
14 unsigned char red = 0, green = 0, blue = 0;
15 unsigned char value() const { return std::max({red, green, blue}); }
16};
17
18struct Histogramme {
19 std::array<double, 256> h{}, H{};
20 template<std::ranges::input_range R>
21 requires std::same_as<std::ranges::range_value_t<R>, unsigned char>
22 void init(R&& valeurs) {
23 h.fill(0); H.fill(0);
24 std::size_t n = 0;
25 for (auto v : valeurs) { ++h[v]; ++n; }
26 if (n == 0) throw std::invalid_argument("sequence vide");
27 double cumul = 0;
28 for (std::size_t i = 0; i < h.size(); ++i) {
29 h[i] /= static_cast<double>(n);
30 H[i] = cumul += h[i];
31 }
32 H.back() = 1;
33 }
34};

36template<std::ranges::input_range R, typename Read, typename Write, typename F>
37void transformValues(R&& pixels, Read read, Write write, F f) {
38 for (auto&& pixel : pixels)
39 write(pixel, f(read(pixel)));
40}
41
42template<typename R>
43concept HistogramInput = requires(Histogramme& h, R& r) { h.init(r); };
44
45int main() {
46 std::vector<Color> img{{10,20,30}, {40,50,60}};
47 auto greenChannel = [](auto& c) -> decltype(auto) { return (c.green); };
48 auto writable = img | std::views::transform(greenChannel);
49 auto readonly = std::as_const(img) | std::views::transform(greenChannel);
50 static_assert(std::same_as<std::ranges::range_reference_t<decltype(writable)>, unsigned char&>);
51 static_assert(std::same_as<std::ranges::range_reference_t<decltype(readonly)>, const unsigned char&>);
52 static_assert(std::indirectly_writable<decltype(writable.begin()), unsigned char>);
53 static_assert(HistogramInput<decltype(writable)>);
54 static_assert(!HistogramInput<std::vector<Color>>);
55 static_assert(!HistogramInput<std::vector<double>>);
56 std::vector<unsigned char> extracted(img.size());
57 std::ranges::copy(readonly, extracted.begin());
58 assert((extracted == std::vector<unsigned char>{20,50}));
59 for (auto& v : writable) v = 0;
60 assert(img[0].green == 0 && img[1].green == 0);
61 assert(img[0].red == 10 && img[1].blue == 60);
62 auto values = std::as_const(img) | std::views::transform([](const Color& c) { return c.value(); });
63 Histogramme H;
64 H.init(values);
65 assert(H.h[30] == 0.5 && H.h[60] == 0.5 && H.H.back() == 1);
66 transformValues(img, [](const Color& c) { return c.red; },
67 [](Color& c, unsigned char v) { c.red = v; },
68 [](unsigned char v) { return static_cast<unsigned char>(v + 1); });
69 assert(img[0].red == 11 && img[1].red == 41);
70 std::vector<unsigned char> gris{1,2};
71 transformValues(gris, [](unsigned char v) { return v; },
72 [](unsigned char& p, unsigned char v) { p = v; },
73 [](unsigned char v) { return static_cast<unsigned char>(2*v); });
74 assert((gris == std::vector<unsigned char>{2,4}));
75}
